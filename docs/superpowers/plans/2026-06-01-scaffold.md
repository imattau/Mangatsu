# Mangatsu Scaffold Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Scaffold a Vite + React + TypeScript project with Tailwind CSS, shadcn/ui, React Router v6, Zustand, Vitest, and Applesauce/Blossom service stubs.

**Architecture:** All Nostr interaction is encapsulated in `NostrService`; all Blossom file operations in `BlossomService`. Zustand stores hold app state and are populated by services. Screens consume stores via hooks and never call services directly.

**Tech Stack:** Vite 5, React 18, TypeScript 5, Tailwind CSS v3, shadcn/ui, React Router v6, Zustand 4, applesauce-core/relay/accounts/signers/factory/loaders/react, blossom-client-sdk, Vitest, React Testing Library

```ts
import { EventStore } from 'applesauce-core'
import { RelayPool } from 'applesauce-relay'
import { AccountManager } from 'applesauce-accounts'
import { EventFactory } from 'applesauce-factory'

const DEFAULT_RELAYS = [
  'wss://relay.damus.io',
  'wss://relay.primal.net',
  'wss://nos.lol',
]

export class NostrService {
  eventStore = new EventStore()
  relayPool = new RelayPool()
  accountManager = new AccountManager()
  eventFactory = new EventFactory({ signer: undefined })

  async connect(relays: string[] = DEFAULT_RELAYS) {
    for (const url of relays) {
      this.relayPool.add(url)
    }
  }

  async disconnect() {
    this.relayPool.close()
  }

  get activeAccount() {
    return this.accountManager.active
  }
}

export const nostrService = new NostrService()
```

- [ ] **Step 2: Create `src/stores/comicStore.ts`**

```ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import type { Comic, Chapter } from '@/types'

interface ComicState {
  comics: Record<string, Comic>
  chapters: Record<string, Chapter>
  setComic: (comic: Comic) => void
  setChapter: (chapter: Chapter) => void
  chaptersForComic: (comicDTag: string) => Chapter[]
}

export const useComicStore = create<ComicState>()(
  persist(
    (set, get) => ({
      comics: {},
      chapters: {},
      setComic: (comic) =>
        set((s) => ({ comics: { ...s.comics, [comic.id]: comic } })),
      setChapter: (chapter) =>
        set((s) => ({ chapters: { ...s.chapters, [chapter.id]: chapter } })),
      chaptersForComic: (comicDTag) =>
        Object.values(get().chapters).filter((c) => c.parentDTag === comicDTag),
    }),
    { name: 'comics' }
  )
)
```

```ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import type { BlossomServer } from '@/types'

interface BlossomState {
  servers: BlossomServer[]
  cachedHashes: Record<string, string>  // hash → local object URL
  setServers: (servers: BlossomServer[]) => void
  setCachedHash: (hash: string, objectUrl: string) => void
  primaryServer: () => string | undefined
}

export const useBlossomStore = create<BlossomState>()(
  persist(
    (set, get) => ({
      servers: [],
      cachedHashes: {},
      setServers: (servers) => set({ servers }),
      setCachedHash: (hash, objectUrl) =>
        set((s) => ({ cachedHashes: { ...s.cachedHashes, [hash]: objectUrl } })),
      primaryServer: () => get().servers[0]?.url,
    }),
    { name: 'blossom' }
  )
)
```
